# Proper combinatorial Laplacian (replace Gram matrix) 
import numpy as np 
from scipy.sparse.linalg import eigsh 
def compute_sheaf_laplacian(adjacency, fiber_dim=5): 
    D = np.diag(adjacency.sum(axis=1)) 
    L = D - adjacency  # combinatorial Laplacian 
    # Fiber sheaf block extension (for F3/F7) 
    # diagonal blocks ← sum of restriction maps per vertex 
    # off-diagonal blocks ← −weight * cross-term for adjacent vertices 
    K_S = build_fiber_sheaf_block_matrix(L, fiber_dim)  # positive semi-definite kernel 
    eigvals = eigsh(K_S, k=2, which='SM')[0] 
    lambda_1 = eigvals[1]  # algebraic connectivity 
    return K_S, lambda_1
